class ScoreTracker {
  constructor(unitSize = 8) {
    const normalized = Math.floor(unitSize);
    this.unitSize = normalized > 0 ? normalized : 1;
    this.highWater = 0;
  }
  reset(initialPosition = 0) {
    this.highWater = Math.max(0, Number.isFinite(initialPosition) ? initialPosition : 0);
  }
  award(currentPosition) {
    const position = Math.max(0, Number.isFinite(currentPosition) ? currentPosition : 0);
    if (position <= this.highWater) return 0;
    const previousUnits = Math.floor(this.highWater / this.unitSize);
    const nextUnits = Math.floor(position / this.unitSize);
    this.highWater = position;
    const delta = nextUnits - previousUnits;
    return delta > 0 ? delta : 0;
  }
  getHighWater() { return this.highWater; }
}

function drawSprite(spriteData, x, y, scale = 1, flipX = false) {
  const s = scale;
  for (let row = 0; row < spriteData.length; row++) {
    for (let col = 0; col < spriteData[row].length; col++) {
      const c = spriteData[row][col];
      if (!c) continue;
      ctx.fillStyle = c;
      const px = flipX ? x + (spriteData[row].length - 1 - col) * s : x + col * s;
      ctx.fillRect(Math.floor(px), Math.floor(y + row * s), s, s);
    }
  }
}

        const py = groundY - 3 - Math.floor(Math.random() * 3);
        for (let px = x; px < Math.min(x + gapW, W - 5); px++) {
          tiles[py][px] = TILE.PLATFORM;
        }
      }
      lastGap = x;
      x += gapW;
      continue;
    }

    if (Math.random() < spikeChance && tiles[groundY][x] === TILE.GRASS) {
      tiles[groundY - 1][x] = TILE.SPIKE;
    }

    if (Math.random() < 0.03) {
      const py = groundY - 4 - Math.floor(Math.random() * 6);
      const pw = 3 + Math.floor(Math.random() * 5);
      for (let px = x; px < Math.min(x + pw, W); px++) {
        if (py >= 2 && py < H) tiles[py][px] = Math.random() < 0.15 ? TILE.BREAKABLE : TILE.GRASS;
      }
      for (let px = x; px < Math.min(x + pw, W); px++) {
        if (Math.random() < 0.4 && py - 1 >= 0) tiles[py - 1][px] = TILE.COIN;
      }
      if (Math.random() < gemChance * 5 && py - 1 >= 0) tiles[py - 1][x + Math.floor(pw / 2)] = TILE.GEM;
    }

    if (Math.random() < coinChance && tiles[groundY][x] === TILE.GRASS) {
      tiles[groundY - 1][x] = TILE.COIN;
    }

    if (Math.random() < gemChance && tiles[groundY][x] === TILE.GRASS) {
      const gy = groundY - 3 - Math.floor(Math.random() * 4);
      if (gy >= 0) tiles[gy][x] = TILE.GEM;
    }

    if (Math.random() < powerUpChance) {
      const py = groundY - 2 - Math.floor(Math.random() * 4);
      if (py >= 0) {
        const types = [TILE.POWERUP_SPEED, TILE.POWERUP_SHIELD, TILE.POWERUP_MAGNET];
        tiles[py][x] = types[Math.floor(Math.random() * types.length)];
      }
    }

    if (Math.random() < enemyChance && tiles[groundY][x] === TILE.GRASS) {
      enemies.push({
        x: x * CONFIG.TILE_SIZE, y: (groundY - 1) * CONFIG.TILE_SIZE,
        type: Math.random() < 0.3 + diff * 0.05 ? 'flyer' : 'walker',
        vx: (Math.random() < 0.5 ? 1 : -1) * (1 + diff * 0.15),
        startX: x * CONFIG.TILE_SIZE,
        patrolRange: (60 + Math.random() * 80),
        frame: 0, alive: true
      });
      if (enemies[enemies.length - 1].type === 'flyer') {
        enemies[enemies.length - 1].y -= (3 + Math.random() * 4) * CONFIG.TILE_SIZE;
        enemies[enemies.length - 1].baseY = enemies[enemies.length - 1].y;
        enemies[enemies.length - 1].flyPhase = Math.random() * Math.PI * 2;
      }
    }

    if (Math.random() < 0.01) {
      movingPlatforms.push({
        x: x * CONFIG.TILE_SIZE, y: (groundY - 5 - Math.floor(Math.random() * 4)) * CONFIG.TILE_SIZE,
        w: (3 + Math.floor(Math.random() * 3)) * CONFIG.TILE_SIZE, h: CONFIG.TILE_SIZE,
        dir: Math.random() < 0.5 ? 'h' : 'v',
        speed: 0.5 + Math.random() * 0.8, phase: Math.random() * Math.PI * 2,
        amplitude: 40 + Math.random() * 60
      });
    }
  }

  tiles[groundY - 1][W - 5] = TILE.EXIT;
  tiles[groundY - 2][W - 5] = TILE.EXIT;

  if (diff > 3) {
    for (let x = 20; x < W - 10; x++) {
      if (Math.random() < 0.01 * diff) {
        for (let ix = x; ix < Math.min(x + 5 + Math.floor(Math.random() * 8), W); ix++) {
          if (tiles[groundY][ix] === TILE.GRASS) tiles[groundY][ix] = TILE.ICE;
        }
      }
    }
  }

  return { tiles, enemies, movingPlatforms, width: W, height: H };
}

// ═══════════════════════════════════════════════
// TILE UTILITIES
// ═══════════════════════════════════════════════
function getTileAt(px, py) {
  const col = Math.floor(px / CONFIG.TILE_SIZE);
  const row = Math.floor(py / CONFIG.TILE_SIZE);
  if (row < 0 || row >= gs.level.height || col < 0 || col >= gs.level.width) return TILE.EMPTY;
  return gs.level.tiles[row][col];
}

function isSolid(tileType) {
  return tileType === TILE.GRASS || tileType === TILE.STONE || tileType === TILE.ICE || tileType === TILE.BREAKABLE;
}

function isOnIce(player) {
  const belowTile = getTileAt(player.x + player.w / 2, player.y + player.h + 1);
  return belowTile === TILE.ICE;
}

// ═══════════════════════════════════════════════
// COLLISION RESOLUTION
// ═══════════════════════════════════════════════
function resolveXCollision(p) {
  const TS = CONFIG.TILE_SIZE;
  const checkYs = [p.y + 1, p.y + p.h / 2, p.y + p.h - 1];

  if (p.vx > 0) {
    const rightEdge = p.x + p.w;
    for (const cy of checkYs) {
      const tile = getTileAt(rightEdge, cy);
      if (isSolid(tile)) {
        const col = Math.floor(rightEdge / TS);
        p.x = col * TS - p.w;
        p.vx = 0;
        break;
      }
    }
  } else if (p.vx < 0) {
    for (const cy of checkYs) {
      const tile = getTileAt(p.x, cy);
      if (isSolid(tile)) {
        const col = Math.floor(p.x / TS);
        p.x = (col + 1) * TS;
        p.vx = 0;
        break;
      }
    }
  }
}

function resolveYCollision(p, oldBottom) {
  const TS = CONFIG.TILE_SIZE;
  const checkXs = [p.x + 1, p.x + p.w / 2, p.x + p.w - 1];

  p.onGround = false;

  if (p.vy > 0) {
    const bottomEdge = p.y + p.h;
    for (const cx of checkXs) {
      const tile = getTileAt(cx, bottomEdge);
      const row = Math.floor(bottomEdge / TS);
      const platformTop = row * TS;

      if (isSolid(tile)) {
        p.y = platformTop - p.h;
        p.vy = 0;
        p.onGround = true;
        break;
      }
      if (tile === TILE.PLATFORM && oldBottom <= platformTop + 2) {
        p.y = platformTop - p.h;
        p.vy = 0;
        p.onGround = true;
        break;
      }
    }
  } else if (p.vy < 0) {
    for (const cx of checkXs) {
      const tile = getTileAt(cx, p.y);
      if (isSolid(tile)) {
        const row = Math.floor(p.y / TS);
        p.y = (row + 1) * TS;
        p.vy = 0;
        if (tile === TILE.BREAKABLE) {
          gs.level.tiles[row][Math.floor(cx / TS)] = TILE.EMPTY;
        }
        break;
      }
    }
  }
}

// ═══════════════════════════════════════════════
// TILE INTERACTIONS
// ═══════════════════════════════════════════════
function checkTilePickups(p) {
  const TS = CONFIG.TILE_SIZE;
  const startCol = Math.floor(p.x / TS);
  const endCol = Math.floor((p.x + p.w - 1) / TS);
  const startRow = Math.floor(p.y / TS);
  const endRow = Math.floor((p.y + p.h - 1) / TS);

  for (let row = startRow; row <= endRow; row++) {
    for (let col = startCol; col <= endCol; col++) {
      if (row < 0 || row >= gs.level.height || col < 0 || col >= gs.level.width) continue;
      const tile = gs.level.tiles[row][col];

      if (tile === TILE.COIN) {
        gs.coins++; gs.score += 10;
        gs.level.tiles[row][col] = TILE.EMPTY;
      } else if (tile === TILE.GEM) {
        gs.gems++; gs.score += 50;
        gs.level.tiles[row][col] = TILE.EMPTY;
      } else if (tile === TILE.POWERUP_SPEED) {
        gs.powerUps.speed = 600;
        gs.level.tiles[row][col] = TILE.EMPTY;
      } else if (tile === TILE.POWERUP_SHIELD) {
        gs.player.shieldHP = gs.player.shieldMax || 1;
        gs.level.tiles[row][col] = TILE.EMPTY;
      } else if (tile === TILE.POWERUP_MAGNET) {
        gs.powerUps.magnet = 600;
        gs.level.tiles[row][col] = TILE.EMPTY;
      }
    }
  }
}

function checkSpikeCollision(p) {
  const TS = CONFIG.TILE_SIZE;
  const checkPoints = [
    [p.x + 2, p.y + p.h - 2],
    [p.x + p.w - 2, p.y + p.h - 2],
    [p.x + p.w / 2, p.y + p.h - 1]
  ];
  for (const [cx, cy] of checkPoints) {
    if (getTileAt(cx, cy) === TILE.SPIKE) {
      if (p.invincible > 0) return;
      if (p.shieldHP > 0) { p.shieldHP--; p.invincible = 60; return; }
      killPlayer();
      return;
    }
  }
}

function checkExitCollision(p) {
  const TS = CONFIG.TILE_SIZE;
  const cx = p.x + p.w / 2;
  const cy = p.y + p.h / 2;
  if (getTileAt(cx, cy) === TILE.EXIT) {
    gs.levelNum++;
    startGame();
  }
}

// ═══════════════════════════════════════════════
// PLAYER UPDATE
// ═══════════════════════════════════════════════
function updatePlayer() {
  const p = gs.player;
  if (!p || !p.alive) return;

  const oldY = p.y;
  const oldBottom = oldY + p.h;

  if (p.invincible > 0) p.invincible--;
  if (gs.powerUps.speed > 0) gs.powerUps.speed--;
  if (gs.powerUps.shield > 0) gs.powerUps.shield--;
  if (gs.powerUps.magnet > 0) gs.powerUps.magnet--;

  const onIce = isOnIce(p);
  const wantLeft = keys['ArrowLeft'] || keys['KeyA'];
  const wantRight = keys['ArrowRight'] || keys['KeyD'];

  let baseSpeed = CONFIG.PLAYER_SPEED;
  if (p.skin && p.skin.bonus === 'speed') baseSpeed *= 1.1;
  if (gs.powerUps.speed > 0) baseSpeed *= 1.3;

  if (onIce) {
    if (wantLeft) { p.vx -= 0.15; p.facing = -1; }
    else if (wantRight) { p.vx += 0.15; p.facing = 1; }
    else { p.vx *= 0.96; }
    p.vx = Math.max(-baseSpeed, Math.min(p.vx, baseSpeed));
    if (Math.abs(p.vx) < 0.1 && !wantLeft && !wantRight) p.vx = 0;
  } else {
    if (wantLeft) { p.vx = -baseSpeed; p.facing = -1; }
    else if (wantRight) { p.vx = baseSpeed; p.facing = 1; }
    else { p.vx = 0; }
  }

  p.vy += CONFIG.GRAVITY;
  if (p.vy > CONFIG.MAX_FALL_SPEED) p.vy = CONFIG.MAX_FALL_SPEED;

  p.x += p.vx;
  p.x = Math.max(0, Math.min(p.x, gs.level.width * CONFIG.TILE_SIZE - p.w));

  resolveXCollision(p);

  p.y += p.vy;

  if (p.y > gs.level.height * CONFIG.TILE_SIZE + 50) {
    killPlayer();
    return;
  }

  const wasAbove = oldBottom;
  resolveYCollision(p, wasAbove);

  if (p.onGround) {
    p.coyoteTimer = CONFIG.COYOTE_FRAMES;
    p.jumpsLeft = p.maxJumps;
  } else {
    if (p.coyoteTimer > 0) p.coyoteTimer--;
  }

  const jumpBuffered = (frameCount - lastJumpPress) < CONFIG.JUMP_BUFFER_FRAMES;
  if (jumpBuffered && (p.onGround || p.coyoteTimer > 0) && p.jumpsLeft > 0) {
    let jumpVel = CONFIG.JUMP_VELOCITY;
    if (p.skin && p.skin.bonus === 'jump') jumpVel *= 1.12;
    p.vy = jumpVel;
    p.onGround = false;
    p.coyoteTimer = 0;
    p.jumpsLeft--;
    lastJumpPress = -100;
  }

  if (p.onGround && Math.abs(p.vx) > 0.5) {
    p.frameTimer++;
    if (p.frameTimer >= 8) { p.frame = (p.frame + 1) % 4; p.frameTimer = 0; }
  } else {
    p.frame = 0; p.frameTimer = 0;
  }

  checkTilePickups(p);
  checkSpikeCollision(p);
  checkExitCollision(p);

  scoreTracker.award(gs.progressBase + p.x + p.w);
  gs.score = scoreTracker.getHighWater();
}

// ═══════════════════════════════════════════════
// ENEMY UPDATES
// ═══════════════════════════════════════════════
function updateEnemies() {
  for (const e of gs.entities) {
    if (!e.alive) {
      if (e.deathTimer !== undefined && e.deathTimer > 0) {
        e.deathTimer--;
      }
      continue;
    }

    if (e.type === 'walker') {
      e.x += e.vx;

      if (e.x < e.startX - e.patrolRange) {
        e.vx = Math.abs(e.vx);
      } else if (e.x > e.startX + e.patrolRange) {
        e.vx = -Math.abs(e.vx);
      }

      const wallCheckX = e.vx > 0 ? e.x + 16 : e.x - 1;
      if (isSolid(getTileAt(wallCheckX, e.y + 8))) {
        e.vx = -e.vx;
      }

      const ledgeCheckX = e.vx > 0 ? e.x + 16 : e.x - 1;
      const tileBelow = getTileAt(ledgeCheckX, e.y + 17);
      if (!isSolid(tileBelow) && tileBelow !== TILE.PLATFORM) {
        e.vx = -e.vx;
      }

      e.frame = (e.frame + 0.1) % 2;

    } else if (e.type === 'flyer') {
      e.flyPhase += 0.03;
      e.y = e.baseY + Math.sin(e.flyPhase) * 20;

      e.x += e.vx * 0.5;
      if (e.x < e.startX - e.patrolRange) {
        e.vx = Math.abs(e.vx);
      } else if (e.x > e.startX + e.patrolRange) {
        e.vx = -Math.abs(e.vx);
      }

      e.frame = (e.frame + 0.08) % 2;
    }
  }

  if (frameCount % 60 === 0) {
    gs.entities = gs.entities.filter(e => e.alive || (e.deathTimer !== undefined && e.deathTimer > 0));
  }
}

// ═══════════════════════════════════════════════
// ENEMY COLLISION CHECK
// ═══════════════════════════════════════════════
function checkEnemyCollision() {
  const p = gs.player;
  if (!p || !p.alive || p.invincible > 0) return;

  for (const e of gs.entities) {
    if (!e.alive) continue;

    const ew = 16, eh = 16;
    if (p.x + p.w > e.x && p.x < e.x + ew &&
        p.y + p.h > e.y && p.y < e.y + eh) {

      const feetOverlap = (p.y + p.h) - e.y;
      if (p.vy > 0 && feetOverlap < 10) {
        e.alive = false;
        e.deathTimer = 10;
        p.vy = CONFIG.JUMP_VELOCITY * 0.6;
        gs.score += 25;
      } else {
        if (p.shieldHP > 0) {
          p.shieldHP--;
          p.invincible = 60;
        } else {
          killPlayer();
        }
      }
      return;
    }
  }
}

  gs.player.alive = false;

}

let _stars = null;
function initStars() {
  _stars = [];
  for (let i = 0; i < 80; i++) {
    _stars.push({
      x: Math.random() * CONFIG.CANVAS_WIDTH * 4,
      y: Math.random() * CONFIG.CANVAS_HEIGHT,
      size: i < 40 ? 1 : (1 + Math.random()),
      color: i < 40 ? '#335' : '#668',
      parallax: i < 40 ? 0.05 : 0.1
    });
  }
}

function renderBackground() {
  ctx.fillStyle = '#0a0a1a';
  ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);
  const grad = ctx.createLinearGradient(0, 0, 0, CONFIG.CANVAS_HEIGHT);
  grad.addColorStop(0, '#0a0a1a');
  grad.addColorStop(1, '#0f1520');
  ctx.fillStyle = grad;
  ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);

  if (!_stars) initStars();
  for (const star of _stars) {
    const screenX = ((star.x - gs.camera.x * star.parallax) % (CONFIG.CANVAS_WIDTH * 4) + CONFIG.CANVAS_WIDTH * 4) % (CONFIG.CANVAS_WIDTH * 4);
    if (screenX < CONFIG.CANVAS_WIDTH) {
      ctx.fillStyle = star.color;
      ctx.fillRect(screenX, star.y, star.size, star.size);
    }
  }
}

function updateCamera() {
  const targetX = gs.player.x - CONFIG.CANVAS_WIDTH / 2 + gs.player.w / 2;
  const targetY = Math.max(0, gs.player.y - CONFIG.CANVAS_HEIGHT * 0.65);
  gs.camera.x += (targetX - gs.camera.x) * CONFIG.CAMERA_LERP;
  gs.camera.x = Math.max(0, Math.min(gs.camera.x, gs.level.width * CONFIG.TILE_SIZE - CONFIG.CANVAS_WIDTH));
  gs.camera.y += (targetY - gs.camera.y) * CONFIG.CAMERA_LERP;
  gs.camera.y = Math.max(0, Math.min(gs.camera.y, Math.max(0, gs.level.height * CONFIG.TILE_SIZE - CONFIG.CANVAS_HEIGHT)));
}

// ═══════════════════════════════════════════════
// RENDERING - TILES
// ═══════════════════════════════════════════════
function renderTiles() {
  const TS = CONFIG.TILE_SIZE;
  const startCol = Math.max(0, Math.floor(gs.camera.x / TS) - 1);
  const endCol = Math.min(gs.level.width, Math.ceil((gs.camera.x + CONFIG.CANVAS_WIDTH) / TS) + 1);
  const startRow = Math.max(0, Math.floor(gs.camera.y / TS) - 1);
  const endRow = Math.min(gs.level.height, Math.ceil((gs.camera.y + CONFIG.CANVAS_HEIGHT) / TS) + 1);

  for (let row = startRow; row < endRow; row++) {
    for (let col = startCol; col < endCol; col++) {
      const tile = gs.level.tiles[row][col];
      if (tile === TILE.EMPTY) continue;

      const sx = col * TS - gs.camera.x;
      const sy = row * TS - gs.camera.y;

      if (tile === TILE.GRASS) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.fillRect(sx, sy, TS, TS);
        ctx.fillStyle = TILE_COLORS[tile][1];
        ctx.fillRect(sx, sy + TS - 2, TS, 2);
        const h = ((col * 7 + row * 13) & 0x7FFFFFFF) % 4;
        ctx.fillStyle = '#6c5';
        ctx.fillRect(sx + h * 4, sy + 2, 2, 2);
      } else if (tile === TILE.STONE) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.fillRect(sx, sy, TS, TS);
        ctx.strokeStyle = TILE_COLORS[tile][1];
        ctx.lineWidth = 1;
        for (let y = sy; y < sy + TS; y += 4) {
          ctx.beginPath();
          ctx.moveTo(sx, y);
          ctx.lineTo(sx + TS, y);
          ctx.stroke();
        }
        for (let x = sx; x < sx + TS; x += 4) {
          ctx.beginPath();
          ctx.moveTo(x, sy);
          ctx.lineTo(x, sy + TS);
          ctx.stroke();
        }
      } else if (tile === TILE.SPIKE) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.beginPath();
        ctx.moveTo(sx + TS / 2, sy);
        ctx.lineTo(sx + TS, sy + TS);
        ctx.lineTo(sx, sy + TS);
        ctx.closePath();
        ctx.fill();
      } else if (tile === TILE.ICE) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.fillRect(sx, sy, TS, TS);
        ctx.fillStyle = '#fff';
        ctx.fillRect(sx + 2, sy + 2, 3, 2);
        ctx.fillRect(sx + 10, sy + 8, 4, 2);
      } else if (tile === TILE.BREAKABLE) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.fillRect(sx, sy, TS, TS);
        ctx.strokeStyle = TILE_COLORS[tile][1];
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(sx + 4, sy + 4);
        ctx.lineTo(sx + 12, sy + 12);
        ctx.stroke();
        ctx.beginPath();
        ctx.moveTo(sx + 12, sy + 4);
        ctx.lineTo(sx + 4, sy + 12);
        ctx.stroke();
      } else if (tile === TILE.PLATFORM) {
        ctx.fillStyle = TILE_COLORS[tile][0];
        ctx.fillRect(sx, sy, TS, TS);
        ctx.fillStyle = '#9c7';
        ctx.fillRect(sx, sy, TS, 1);
        ctx.fillStyle = TILE_COLORS[tile][1];
        ctx.fillRect(sx, sy + TS - 1, TS, 1);
      } else if (tile === TILE.COIN) {
        const bobY = sy + Math.sin(frameCount * 0.08 + col) * 2;
        ctx.fillStyle = '#ff0';
        ctx.beginPath();
        ctx.arc(sx + TS / 2, bobY + TS / 2, 3, 0, Math.PI * 2);
        ctx.fill();
      } else if (tile === TILE.GEM) {
        ctx.fillStyle = '#4ef';
        ctx.beginPath();
        ctx.moveTo(sx + TS / 2, sy + 2);
        ctx.lineTo(sx + TS - 2, sy + TS / 2);
        ctx.lineTo(sx + TS / 2, sy + TS - 2);
        ctx.lineTo(sx + 2, sy + TS / 2);
        ctx.closePath();
        ctx.fill();
        ctx.strokeStyle = '#2cd';
        ctx.lineWidth = 1;
        ctx.stroke();
      } else if (tile === TILE.POWERUP_SPEED) {
        const scale = 1 + Math.sin(frameCount * 0.06) * 0.1;
        const r = 4 * scale;
        ctx.fillStyle = '#4f4';
        ctx.beginPath();
        ctx.arc(sx + TS / 2, sy + TS / 2, r, 0, Math.PI * 2);
        ctx.fill();
      } else if (tile === TILE.POWERUP_SHIELD) {
        const scale = 1 + Math.sin(frameCount * 0.06) * 0.1;
        const r = 4 * scale;
        ctx.fillStyle = '#48f';
        ctx.beginPath();
        ctx.arc(sx + TS / 2, sy + TS / 2, r, 0, Math.PI * 2);
        ctx.fill();
      } else if (tile === TILE.POWERUP_MAGNET) {
        const scale = 1 + Math.sin(frameCount * 0.06) * 0.1;
        const r = 4 * scale;
        ctx.fillStyle = '#f4f';
        ctx.beginPath();
        ctx.arc(sx + TS / 2, sy + TS / 2, r, 0, Math.PI * 2);
        ctx.fill();
      } else if (tile === TILE.EXIT) {
        const pulse = Math.sin(frameCount * 0.04);
        ctx.fillStyle = pulse > 0 ? '#ff0' : '#fc0';
        ctx.fillRect(sx + 2, sy + 2, TS - 4, TS - 4);
        ctx.strokeStyle = pulse > 0 ? '#fc0' : '#ff0';
        ctx.lineWidth = 1;
        ctx.strokeRect(sx + 2, sy + 2, TS - 4, TS - 4);

// ═══════════════════════════════════════════════
// RENDERING - MOVING PLATFORMS
// ═══════════════════════════════════════════════
function renderMovingPlatforms() {
  for (const p of gs.level.movingPlatforms) {
    let vx = p.x, vy = p.y;
    if (p.dir === 'h') {
      vx += Math.sin(frameCount * p.speed * 0.02 + p.phase) * p.amplitude;
    } else if (p.dir === 'v') {
      vy += Math.sin(frameCount * p.speed * 0.02 + p.phase) * p.amplitude;
    }

    const px = vx - gs.camera.x;
    const py = vy - gs.camera.y;

    ctx.fillStyle = TILE_COLORS[TILE.PLATFORM][0];
    ctx.fillRect(px, py, p.w, p.h);
    ctx.fillStyle = '#9c7';
    ctx.fillRect(px, py, p.w, 1);
    ctx.fillStyle = TILE_COLORS[TILE.PLATFORM][1];
    ctx.fillRect(px, py + p.h - 1, p.w, 1);
  }
}

// RENDERING - ENTITIES

function renderEntities() {
  for (const e of gs.entities) {
    if (!e.alive && (e.deathTimer === undefined || e.deathTimer <= 0)) continue;

    if (!e.alive && e.deathTimer > 0) {
      if (e.deathTimer % 2 === 0) continue;
    }

    const ex = e.x - gs.camera.x;
    const ey = e.y - gs.camera.y;

    if (e.type === 'walker') {
      ctx.fillStyle = '#f44';
      ctx.fillRect(ex, ey, 16, 8);
      ctx.fillStyle = '#fff';
      ctx.fillRect(ex + 3, ey + 2, 2, 2);
      ctx.fillRect(ex + 11, ey + 2, 2, 2);
      ctx.fillStyle = '#d22';
      ctx.fillRect(ex + 2, ey + 8, 3, 4);
      ctx.fillRect(ex + 11, ey + 8, 3, 4);
    } else if (e.type === 'flyer') {
      ctx.fillStyle = '#85f';
      ctx.fillRect(ex + 3, ey + 2, 10, 8);
      const wingL = Math.sin(frameCount * 0.15) * 3;
      const wingR = Math.sin(frameCount * 0.15 + Math.PI) * 3;
      ctx.fillStyle = '#a6f';
      ctx.fillRect(ex + 1, ey + 3 + wingL, 2, 4);
      ctx.fillRect(ex + 13, ey + 3 + wingR, 2, 4);
    }
  }
}

function renderPlayer() {
  const p = gs.player;

  if (p.invincible > 0 && frameCount % 4 < 2) return;

  const sprite = makePlayerSprite(p.skin.colors, p.frame);

    ctx.fillStyle = '#4ef';
    ctx.beginPath();
    ctx.arc(p.x - gs.camera.x + p.w / 2, p.y - gs.camera.y + p.h / 2, 20, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }
}

- **CLI mode**: Routes all LLM calls through `claude --print` CLI. Pipeline runs fully automated.
- **App mode**: A hybrid mode. Only steps routed to **OpenAI** (architect, synthesize, followups, pr-draft) go through a **prompt queue** — the user manually copies each prompt into the ChatGPT desktop app, pastes the response back into the dashboard inline at the active pipeline step, and submits. **Gemini** steps (critique) run via **Gemini API** automatically. **Claude/Codex** steps (execute) run via **CLI** automatically. - **Mock mode**: Uses fixture files for testing. No LLM calls.

### APP Mode Routing Summary
| Provider | Steps (default) | APP Mode Behavior |
|----------|-----------------|-------------------|
| OpenAI | architect, synthesize, followups, pr-draft | Manual prompt queue (ChatGPT) |
| Gemini | critique | Automatic via Gemini API |
| Codex | execute (feature-lane) | Automatic via CLI |
| Claude | execute (bug/danger-lane) | Automatic via CLI |

### APP Mode UI Flow
1. User clicks "Run Cascade" on a task
2. Pipeline reaches an **OpenAI** step (e.g. architect) → `callLLMApp()` writes `.prompt.md` + `.meta.json` to `state/prompts-queue/` and starts polling for `.response.md`
3. Dashboard auto-refreshes, sees the pending prompt, and renders an **inline prompt/response widget** directly under that pipeline step
4. User copies prompt → pastes into ChatGPT desktop app → copies ChatGPT's answer → pastes into the response textarea → clicks Submit → field locks
5. Server writes `.response.md` → poll loop picks it up → pipeline continues to next step
6. If next step is also OpenAI (e.g. synthesize), a new inline widget appears at that step
7. If next step is Gemini (critique) or CLI (execute), it runs **automatically** without user intervention
8. Unlock button allows editing a locked response before the pipeline picks it up
