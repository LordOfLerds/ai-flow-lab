function isValidEmailShape(email) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
}

function validateLoginInput(email, password) {
  const errors = {};
  const normalizedEmail = email.trim();

  if (!normalizedEmail) {
    errors.email = AUTH_MESSAGES.requiredEmail;
  } else if (!isValidEmailShape(normalizedEmail)) {
    errors.email = AUTH_MESSAGES.invalidEmail;
  }

  if (!password) {
    errors.password = AUTH_MESSAGES.requiredPassword;
  }

  return { errors, normalizedEmail };
}

function drawSprite(spriteData, x, y, scale = 1, flipX = false) {
  const s = scale;
  for (let row = 0; row < spriteData.length; row++) {
    for (let col = 0; col < spriteData[row].length; col++) {
      const c = spriteData[row][col];
      if (!c) continue;
      ctx.fillStyle = c;
      const px = flipX ? x + (spriteData[row].length - 1 - col) * s : x + col * s;
      ctx.fillRect(Math.floor(px), Math.floor(y + row * s), s, s);
    }
  }
}

function generateLevel(levelNum) {
  const W = CONFIG.LEVEL_WIDTH_TILES;
  const H = CONFIG.LEVEL_HEIGHT_TILES;
  const tiles = Array.from({ length: H }, () => new Array(W).fill(TILE.EMPTY));
  const enemies = [];
  const movingPlatforms = [];

  // Ground
  const groundY = H - 2;
  for (let x = 0; x < W; x++) {
    tiles[groundY][x] = TILE.GRASS;
    tiles[groundY + 1][x] = TILE.STONE;
  }

  // Difficulty scaling
  const diff = Math.min(levelNum, 10);
  const gapChance = 0.02 + diff * 0.008;
  const spikeChance = 0.01 + diff * 0.005;
  const enemyChance = 0.008 + diff * 0.003;
  const coinChance = 0.04;
  const gemChance = 0.005 + diff * 0.002;
  const powerUpChance = 0.003;

  // Generate terrain with gaps, platforms, obstacles
  let lastGap = 0;
  for (let x = 8; x < W - 8; x++) {
    // Gaps in ground
    if (Math.random() < gapChance && x - lastGap > 8) {
      const gapW = 3 + Math.floor(Math.random() * (1 + diff * 0.3));
      for (let gx = x; gx < Math.min(x + gapW, W - 5); gx++) {
        tiles[groundY][gx] = TILE.EMPTY;
        tiles[groundY + 1][gx] = TILE.EMPTY;
      }
      // Platform over gap
      if (Math.random() < 0.7) {
        const py = groundY - 3 - Math.floor(Math.random() * 3);
        for (let px = x; px < Math.min(x + gapW, W - 5); px++) {
          tiles[py][px] = TILE.PLATFORM;
        }
      }
      lastGap = x;
      x += gapW;
      continue;
    }

    // Spikes
    if (Math.random() < spikeChance && tiles[groundY][x] === TILE.GRASS) {
      tiles[groundY - 1][x] = TILE.SPIKE;
    }

    // Elevated platforms
    if (Math.random() < 0.03) {
      const py = groundY - 4 - Math.floor(Math.random() * 6);
      const pw = 3 + Math.floor(Math.random() * 5);
      for (let px = x; px < Math.min(x + pw, W); px++) {
        if (py >= 2 && py < H) tiles[py][px] = Math.random() < 0.15 ? TILE.BREAKABLE : TILE.GRASS;
      }
      // Coins on platform
      for (let px = x; px < Math.min(x + pw, W); px++) {
        if (Math.random() < 0.4 && py - 1 >= 0) tiles[py - 1][px] = TILE.COIN;
      }
      // Gem on platform
      if (Math.random() < gemChance * 5 && py - 1 >= 0) tiles[py - 1][x + Math.floor(pw / 2)] = TILE.GEM;
    }

    // Ground coins
    if (Math.random() < coinChance && tiles[groundY][x] === TILE.GRASS) {
      tiles[groundY - 1][x] = TILE.COIN;
    }

    // Gems
    if (Math.random() < gemChance && tiles[groundY][x] === TILE.GRASS) {
      const gy = groundY - 3 - Math.floor(Math.random() * 4);
      if (gy >= 0) tiles[gy][x] = TILE.GEM;
    }

    // Power-ups
    if (Math.random() < powerUpChance) {
      const py = groundY - 2 - Math.floor(Math.random() * 4);
      if (py >= 0) {
        const types = [TILE.POWERUP_SPEED, TILE.POWERUP_SHIELD, TILE.POWERUP_MAGNET];
        tiles[py][x] = types[Math.floor(Math.random() * types.length)];
      }
    }

    // Enemies
    if (Math.random() < enemyChance && tiles[groundY][x] === TILE.GRASS) {
      enemies.push({
        x: x * CONFIG.TILE_SIZE, y: (groundY - 1) * CONFIG.TILE_SIZE,
        type: Math.random() < 0.3 + diff * 0.05 ? 'flyer' : 'walker',
        vx: (Math.random() < 0.5 ? 1 : -1) * (1 + diff * 0.15),
        startX: x * CONFIG.TILE_SIZE,
        patrolRange: (60 + Math.random() * 80),
        frame: 0, alive: true
      });
      if (enemies[enemies.length - 1].type === 'flyer') {
        enemies[enemies.length - 1].y -= (3 + Math.random() * 4) * CONFIG.TILE_SIZE;
        enemies[enemies.length - 1].baseY = enemies[enemies.length - 1].y;
        enemies[enemies.length - 1].flyPhase = Math.random() * Math.PI * 2;
      }
    }

    // Moving platforms
    if (Math.random() < 0.01) {
      movingPlatforms.push({
        x: x * CONFIG.TILE_SIZE, y: (groundY - 5 - Math.floor(Math.random() * 4)) * CONFIG.TILE_SIZE,
        w: (3 + Math.floor(Math.random() * 3)) * CONFIG.TILE_SIZE, h: CONFIG.TILE_SIZE,
        dir: Math.random() < 0.5 ? 'h' : 'v',
        speed: 0.5 + Math.random() * 0.8, phase: Math.random() * Math.PI * 2,
        amplitude: 40 + Math.random() * 60
      });
    }
  }

  // Exit at end
  tiles[groundY - 1][W - 5] = TILE.EXIT;
  tiles[groundY - 2][W - 5] = TILE.EXIT;

  // Ice patches at higher levels
  if (diff > 3) {
    for (let x = 20; x < W - 10; x++) {
      if (Math.random() < 0.01 * diff) {
        for (let ix = x; ix < Math.min(x + 5 + Math.floor(Math.random() * 8), W); ix++) {
          if (tiles[groundY][ix] === TILE.GRASS) tiles[groundY][ix] = TILE.ICE;
        }
      }
    }
  }

  return { tiles, enemies, movingPlatforms, width: W, height: H };
}

  const skin = SKINS[gs.selectedSkin];

    alive: true, invincible: 0,

    shieldHP: 0, shieldMax: 0,

  };

// ═══════════════════════════════════════════════
// GAME INIT
// ═══════════════════════════════════════════════
function startGame() {
  if (!gs.auth.isLoggedIn) {
    renderLogin();
    return;
  }

  gs.level = generateLevel(gs.levelNum);
  gs.player = createPlayer();
  gs.entities = [...gs.level.enemies];
  gs.particles = [];
  gs.floatingTexts = [];
  gs.powerUps = { speed: 0, shield: 0, magnet: 0 };
  gs.camera = { x: 0, y: 0 };
  gs.shakeFrames = 0;
  gs.levelTime = 0;

  // Apply skills
  const hasDoubleJump = gs.unlockedSkills.includes('double_jump');
  const hasShield = gs.unlockedSkills.includes('shield');
  gs.player.maxJumps = hasDoubleJump ? 2 : 1;
  gs.player.jumpsLeft = gs.player.maxJumps;
  if (hasShield) {
    gs.player.shieldHP = 1;
    gs.player.shieldMax = 1;
    // Ghost skin extends shield
    if (SKINS[gs.selectedSkin].bonus === 'shield_dur') gs.player.shieldMax = 2;
  }

  gs.phase = 'PLAYING';
  overlay.classList.add('hidden');
}

// ═══════════════════════════════════════════════
// UPDATE
// ═══════════════════════════════════════════════
function update() {
  if (gs.phase !== 'PLAYING') return;
  frameCount++;
  gs.levelTime++;

  const p = gs.player;
  if (!p.alive) return;

  const T = CONFIG.TILE_SIZE;
  const skin = SKINS[gs.selectedSkin];
  let speedMult = 1;
  if (skin.bonus === 'speed') speedMult *= 1.1;
  if (gs.powerUps.speed > 0) { speedMult *= 1.5; gs.powerUps.speed--; }
  let jumpMult = 1;
  if (skin.bonus === 'jump') jumpMult = 1.12;

  // ── Dash ──
  if (p.dashTimer > 0) {
    p.dashTimer--;
    let dashSpeed = CONFIG.DASH_SPEED;
    if (skin.bonus === 'dash') dashSpeed *= 1.3;
    p.vx = p.dashDir * dashSpeed;
    p.vy = 0;
  } else {
    // ── Horizontal movement ──
    let moveDir = 0;
    if (keys['ArrowLeft'] || keys['KeyA']) moveDir = -1;
    if (keys['ArrowRight'] || keys['KeyD']) moveDir = 1;

    const accel = p.onGround ? 0.8 : 0.4;
    const friction = p.onGround ? 0.75 : 0.9;
    // Ice friction
    const onIce = isOnTile(p, TILE.ICE);
    const fric = onIce ? 0.97 : friction;

    if (moveDir !== 0) {
      p.vx += moveDir * accel;
      p.facing = moveDir;
    } else {
      p.vx *= fric;
      if (Math.abs(p.vx) < 0.2) p.vx = 0;
    }
    p.vx = Math.max(-CONFIG.PLAYER_SPEED * speedMult, Math.min(CONFIG.PLAYER_SPEED * speedMult, p.vx));

    // ── Gravity ──
    p.vy += CONFIG.GRAVITY;
    if (p.vy > CONFIG.MAX_FALL_SPEED) p.vy = CONFIG.MAX_FALL_SPEED;
  }

  // ── Coyote time ──
  if (p.onGround) { p.coyoteTimer = CONFIG.COYOTE_FRAMES; p.jumpsLeft = p.maxJumps; p.canDash = true; }
  else { p.coyoteTimer = Math.max(0, p.coyoteTimer - 1); }

  // ── Jump ──
  const jumpPressed = (frameCount - lastJumpPress) <= CONFIG.JUMP_BUFFER_FRAMES;
  if (jumpPressed) {
    if (p.coyoteTimer > 0 && p.jumpsLeft > 0) {
      p.vy = CONFIG.JUMP_VELOCITY * jumpMult;
      p.onGround = false;
      p.coyoteTimer = 0;
      p.jumpsLeft--;
      lastJumpPress = -100;
      spawnParticles(p.x + p.w / 2, p.y + p.h, '#fff', 5);
    } else if (p.jumpsLeft > 0 && !p.onGround && p.coyoteTimer === 0) {
      // Double jump
      p.vy = CONFIG.JUMP_VELOCITY * jumpMult * 0.85;
      p.jumpsLeft--;
      lastJumpPress = -100;
      spawnParticles(p.x + p.w / 2, p.y + p.h, '#8cf', 8);
    }
  }

  // Variable jump height
  if (!(keys['Space'] || keys['ArrowUp'] || keys['KeyW']) && p.vy < -3) {
    p.vy *= 0.6;
  }

  // ── Dash trigger ──
  if ((frameCount - lastDashPress) <= 2 && p.canDash && gs.unlockedSkills.includes('dash') && p.dashTimer === 0) {
    p.dashTimer = CONFIG.DASH_FRAMES;
    p.dashDir = p.facing;
    p.canDash = false;
    lastDashPress = -100;
    spawnParticles(p.x + p.w / 2, p.y + p.h / 2, '#ff8', 10);
  }

  // ── Move X ──
  p.x += p.vx;
  resolveCollisionX(p);

  // ── Move Y ──
  p.y += p.vy;
  p.onGround = false;
  resolveCollisionY(p);

  // Moving platforms
  for (const mp of gs.level.movingPlatforms) {
    const oldX = mp.x, oldY = mp.y;
    if (mp.dir === 'h') mp.x = mp.x + Math.sin(frameCount * 0.02 * mp.speed + mp.phase) * mp.speed;
    else mp.y = mp.y + Math.sin(frameCount * 0.02 * mp.speed + mp.phase) * mp.speed;

    // Check if player is on this platform
    if (p.y + p.h >= mp.y && p.y + p.h <= mp.y + 8 &&
        p.x + p.w > mp.x && p.x < mp.x + mp.w && p.vy >= 0) {
      p.onGround = true;
      p.y = mp.y - p.h;
      p.vy = 0;
      p.x += mp.x - oldX;
    }
  }

  // ── Clamp to level ──
  if (p.x < 0) p.x = 0;
  if (p.x > (gs.level.width * T - p.w)) p.x = gs.level.width * T - p.w;

  // ── Fall death ──
  if (p.y > gs.level.height * T + 50) {
    playerDie();
    return;
  }

  // ── Tile interactions ──
  const cx = Math.floor((p.x + p.w / 2) / T);
  const cy = Math.floor((p.y + p.h / 2) / T);
  for (let dy = -1; dy <= 1; dy++) {
    for (let dx = -1; dx <= 1; dx++) {
      const tx = cx + dx, ty = cy + dy;
      if (tx < 0 || ty < 0 || tx >= gs.level.width || ty >= gs.level.height) continue;
      const tile = gs.level.tiles[ty][tx];
      const tileBox = { x: tx * T, y: ty * T, w: T, h: T };

      if (!aabbOverlap(p, tileBox)) continue;

      // Spike
      if (tile === TILE.SPIKE) {
        if (p.invincible <= 0) {
          if (p.shieldHP > 0) { p.shieldHP--; p.invincible = 30; spawnParticles(p.x + p.w/2, p.y, '#4cf', 12); }
          else { playerDie(); return; }
        }
      }
      // Coin
      else if (tile === TILE.COIN) {
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        gs.coins++;
        let xpGain = 10;
        if (SKINS[gs.selectedSkin].bonus === 'xp') xpGain = Math.floor(xpGain * 1.15);
        addXP(xpGain);
        gs.score += 50;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#fc4', 6);
        spawnFloatingText(tx * T, ty * T, '+10 XP', '#fc4');
      }
      // Gem
      else if (tile === TILE.GEM) {
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        gs.gems++;
        let xpGain = 50;
        if (SKINS[gs.selectedSkin].bonus === 'xp') xpGain = Math.floor(xpGain * 1.15);
        addXP(xpGain);
        gs.score += 200;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#f4f', 10);
        spawnFloatingText(tx * T, ty * T, '+50 XP', '#f4f');
      }
      // Power-ups
      else if (tile === TILE.POWERUP_SPEED) {
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        gs.powerUps.speed = 480;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#ff4', 12);
        spawnFloatingText(tx * T, ty * T, 'SPEED!', '#ff4');
      }
      else if (tile === TILE.POWERUP_SHIELD) {
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        gs.powerUps.shield = 600;
        p.shieldHP = p.shieldMax || 1;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#4cf', 12);
        spawnFloatingText(tx * T, ty * T, 'SHIELD!', '#4cf');
      }
      else if (tile === TILE.POWERUP_MAGNET) {
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        gs.powerUps.magnet = 480;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#f84', 12);
        spawnFloatingText(tx * T, ty * T, 'MAGNET!', '#f84');
      }
      // Exit
      else if (tile === TILE.EXIT) {
        nextLevel();
        return;
      }
      // Breakable
      else if (tile === TILE.BREAKABLE && p.vy > 0 && p.y + p.h <= ty * T + 4) {
        // Mark as breaking
        gs.level.tiles[ty][tx] = TILE.EMPTY;
        spawnParticles(tx * T + T/2, ty * T + T/2, '#a86', 8);
      }
    }
  }

  // ── Magnet effect ──
  if (gs.powerUps.magnet > 0) {
    gs.powerUps.magnet--;
    magnetPull(p);
  }

  // ── Shield timer ──
  if (gs.powerUps.shield > 0) gs.powerUps.shield--;

  // ── Invincibility ──
  if (p.invincible > 0) p.invincible--;

  // ── Enemy update ──
  for (const e of gs.entities) {
    if (!e.alive) continue;
    e.frame++;

    if (e.type === 'walker') {
      e.x += e.vx;
      if (Math.abs(e.x - e.startX) > e.patrolRange) e.vx *= -1;
      // Ground check
      const ex = Math.floor(e.x / T), ey = Math.floor((e.y + T) / T);
      if (ey < gs.level.height && ex >= 0 && ex < gs.level.width) {
        if (gs.level.tiles[ey][ex] === TILE.EMPTY) e.vx *= -1;
      }
    } else if (e.type === 'flyer') {
      e.x += e.vx;
      e.flyPhase += 0.04;
      e.y = e.baseY + Math.sin(e.flyPhase) * 30;
      if (Math.abs(e.x - e.startX) > e.patrolRange) e.vx *= -1;
    }

    // Collision with player
    const eBox = { x: e.x, y: e.y, w: T, h: T };
    if (aabbOverlap(p, eBox) && p.invincible <= 0) {
      // Stomp from above
      if (p.vy > 0 && p.y + p.h < e.y + T * 0.6) {
        e.alive = false;
        p.vy = CONFIG.JUMP_VELOCITY * 0.6;
        gs.score += 100;
        spawnParticles(e.x + T/2, e.y + T/2, '#f44', 10);
        spawnFloatingText(e.x, e.y, '+100', '#f44');
        shake(3, 3);
      } else {
        // Take damage
        if (p.shieldHP > 0) { p.shieldHP--; p.invincible = 30; spawnParticles(p.x + p.w/2, p.y, '#4cf', 12); }
        else if (p.dashTimer > 0) { e.alive = false; gs.score += 100; spawnParticles(e.x + T/2, e.y, '#ff8', 10); }
        else { playerDie(); return; }
      }
    }
  }

  // ── Animation ──
  p.frameTimer++;
  if (p.frameTimer > 6) { p.frameTimer = 0; p.frame = (p.frame + 1) % 4; }

  // ── Camera ──
  const targetX = p.x - CONFIG.CANVAS_WIDTH / 2 + p.w / 2;
  const targetY = p.y - CONFIG.CANVAS_HEIGHT / 2 + p.h / 2;
  gs.camera.x += (targetX - gs.camera.x) * CONFIG.CAMERA_LERP;
  gs.camera.y += (targetY - gs.camera.y) * CONFIG.CAMERA_LERP;
  gs.camera.x = Math.max(0, Math.min(gs.level.width * T - CONFIG.CANVAS_WIDTH, gs.camera.x));
  gs.camera.y = Math.max(0, Math.min(gs.level.height * T - CONFIG.CANVAS_HEIGHT, gs.camera.y));

  // ── Particles ──
  updateParticles();
  updateFloatingTexts();

  // ── Score ──
  gs.score += 1;

  // ── Shake ──
  if (gs.shakeFrames > 0) gs.shakeFrames--;

  // ── HUD update ──
  updateHUD();
}

// ═══════════════════════════════════════════════
// COLLISION HELPERS
// ═══════════════════════════════════════════════
function aabbOverlap(a, b) {
  return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
}

function isSolid(tx, ty) {
  if (tx < 0 || ty < 0 || tx >= gs.level.width || ty >= gs.level.height) return false;
  const t = gs.level.tiles[ty][tx];
  return t === TILE.GRASS || t === TILE.STONE || t === TILE.ICE || t === TILE.BREAKABLE || t === TILE.PLATFORM;
}

function isOnTile(p, tileType) {
  const T = CONFIG.TILE_SIZE;
  const ty = Math.floor((p.y + p.h + 1) / T);
  for (let dx = 0; dx <= 1; dx++) {
    const tx = Math.floor((p.x + dx * (p.w - 1)) / T);
    if (tx >= 0 && ty >= 0 && tx < gs.level.width && ty < gs.level.height) {
      if (gs.level.tiles[ty][tx] === tileType) return true;
    }
  }
  return false;
}

function resolveCollisionX(p) {
  const T = CONFIG.TILE_SIZE;
  const top = Math.floor(p.y / T), bot = Math.floor((p.y + p.h - 1) / T);
  if (p.vx > 0) {
    const tx = Math.floor((p.x + p.w) / T);
    for (let ty = top; ty <= bot; ty++) {
      if (isSolid(tx, ty)) { p.x = tx * T - p.w; p.vx = 0; return; }
    }
  } else if (p.vx < 0) {
    const tx = Math.floor(p.x / T);
    for (let ty = top; ty <= bot; ty++) {
      if (isSolid(tx, ty)) { p.x = (tx + 1) * T; p.vx = 0; return; }
    }
  }
}

function resolveCollisionY(p) {
  const T = CONFIG.TILE_SIZE;
  const left = Math.floor(p.x / T), right = Math.floor((p.x + p.w - 1) / T);
  if (p.vy > 0) {
    const ty = Math.floor((p.y + p.h) / T);
    for (let tx = left; tx <= right; tx++) {
      if (isSolid(tx, ty)) { p.y = ty * T - p.h; p.vy = 0; p.onGround = true; return; }
    }
  } else if (p.vy < 0) {
    const ty = Math.floor(p.y / T);
    for (let tx = left; tx <= right; tx++) {
      if (isSolid(tx, ty) && gs.level.tiles[ty][tx] !== TILE.PLATFORM) { p.y = (ty + 1) * T; p.vy = 0; return; }
    }
  }
}

// ═══════════════════════════════════════════════
// MAGNET PULL
// ═══════════════════════════════════════════════
function magnetPull(p) {
  const T = CONFIG.TILE_SIZE;
  const range = 5 * T;
  const px = p.x + p.w / 2, py = p.y + p.h / 2;
  for (let ty = 0; ty < gs.level.height; ty++) {
    for (let tx = 0; tx < gs.level.width; tx++) {
      const tile = gs.level.tiles[ty][tx];
      if (tile !== TILE.COIN && tile !== TILE.GEM) continue;
      const cx = tx * T + T / 2, cy2 = ty * T + T / 2;
      const dist = Math.sqrt((px - cx) ** 2 + (py - cy2) ** 2);
      if (dist < range && dist > T) {
        // Move coin toward player
        const angle = Math.atan2(py - cy2, px - cx);
        const speed = 3;
        const newX = Math.floor((cx + Math.cos(angle) * speed) / T);
        const newY = Math.floor((cy2 + Math.sin(angle) * speed) / T);
        if (newX !== tx || newY !== ty) {
          if (newX >= 0 && newY >= 0 && newX < gs.level.width && newY < gs.level.height && gs.level.tiles[newY][newX] === TILE.EMPTY) {
            gs.level.tiles[newY][newX] = tile;
            gs.level.tiles[ty][tx] = TILE.EMPTY;
          }
        }
      }
    }
  }
}

// ═══════════════════════════════════════════════
// PARTICLES & EFFECTS
// ═══════════════════════════════════════════════
function spawnParticles(x, y, color, count) {
  for (let i = 0; i < count; i++) {
    gs.particles.push({
      x, y, vx: (Math.random() - 0.5) * 4, vy: (Math.random() - 1) * 3,
      life: 20 + Math.random() * 20, maxLife: 40, color, size: 2 + Math.random() * 2
    });
  }
}

function spawnFloatingText(x, y, text, color) {
  gs.floatingTexts.push({ x, y, text, color, life: 40, maxLife: 40 });
}

function updateParticles() {
  for (let i = gs.particles.length - 1; i >= 0; i--) {
    const p = gs.particles[i];
    p.x += p.vx; p.y += p.vy; p.vy += 0.1; p.life--;
    if (p.life <= 0) gs.particles.splice(i, 1);
  }
}

function updateFloatingTexts() {
  for (let i = gs.floatingTexts.length - 1; i >= 0; i--) {
    const ft = gs.floatingTexts[i];
    ft.y -= 0.8; ft.life--;
    if (ft.life <= 0) gs.floatingTexts.splice(i, 1);
  }
}

function shake(frames, intensity) { gs.shakeFrames = frames; gs.shakeIntensity = intensity; }

// ═══════════════════════════════════════════════
// XP & LEVELING
// ═══════════════════════════════════════════════
function addXP(amount) {
  gs.xp += amount;
  while (gs.xp >= xpForLevel(gs.playerLevel)) {
    gs.xp -= xpForLevel(gs.playerLevel);
    gs.playerLevel++;
    // Check skill unlocks
    for (const skill of SKILLS) {
      if (skill.level <= gs.playerLevel && !gs.unlockedSkills.includes(skill.id)) {
        gs.unlockedSkills.push(skill.id);
        spawnFloatingText(gs.player.x, gs.player.y - 20, 'SKILL: ' + skill.name + '!', '#4f4');
        shake(5, 4);
        // Apply skill immediately
        if (skill.id === 'double_jump') gs.player.maxJumps = 2;
        if (skill.id === 'shield') { gs.player.shieldHP = 1; gs.player.shieldMax = 1; }
      }
    }
    spawnFloatingText(gs.player.x, gs.player.y - 10, 'LEVEL UP!', '#ff4');
    shake(4, 3);
  }
}

function playerDie() {
  const p = gs.player;
  p.alive = false;
  gs.deathCount++;
  shake(6, 5);
  spawnParticles(p.x + p.w/2, p.y + p.h/2, '#f44', 20);

    showGameOver();
  }, 800);
}

// RENDER

function render() {
  ctx.fillStyle = '#0f0f23';
  ctx.fillRect(0, 0, CONFIG.CANVAS_WIDTH, CONFIG.CANVAS_HEIGHT);

  if (gs.phase !== 'PLAYING' && gs.phase !== 'GAME_OVER') return;

  ctx.save();
  // Screen shake
  let sx = 0, sy = 0;
  if (gs.shakeFrames > 0) {
    sx = (Math.random() - 0.5) * gs.shakeIntensity * 2;
    sy = (Math.random() - 0.5) * gs.shakeIntensity * 2;
  }
  ctx.translate(-Math.floor(gs.camera.x) + sx, -Math.floor(gs.camera.y) + sy);

  const T = CONFIG.TILE_SIZE;
  const startTX = Math.max(0, Math.floor(gs.camera.x / T) - 1);
  const endTX = Math.min(gs.level.width, Math.ceil((gs.camera.x + CONFIG.CANVAS_WIDTH) / T) + 1);
  const startTY = Math.max(0, Math.floor(gs.camera.y / T) - 1);
  const endTY = Math.min(gs.level.height, Math.ceil((gs.camera.y + CONFIG.CANVAS_HEIGHT) / T) + 1);

  // ── Background parallax ──
  ctx.fillStyle = '#151530';
  for (let i = 0; i < 60; i++) {
    const bx = (i * 137) % (gs.level.width * T);
    const by = (i * 89) % (CONFIG.CANVAS_HEIGHT * 0.6);
    ctx.globalAlpha = 0.3;
    ctx.fillRect(bx - gs.camera.x * 0.3, by, 2, 2);
  }
  ctx.globalAlpha = 1;

  // ── Tiles ──
  for (let ty = startTY; ty < endTY; ty++) {
    for (let tx = startTX; tx < endTX; tx++) {
      const tile = gs.level.tiles[ty][tx];
      if (tile === TILE.EMPTY) continue;

      const x = tx * T, y = ty * T;

      if (TILE_COLORS[tile]) {
        const [c1, c2] = TILE_COLORS[tile];
        ctx.fillStyle = c1;
        ctx.fillRect(x, y, T, T);
        // Pixel detail
        ctx.fillStyle = c2;
        ctx.fillRect(x, y + T - 3, T, 3);
        if (tile === TILE.SPIKE) {
          // Triangle spike
          ctx.fillStyle = '#f44';
          ctx.beginPath();
          ctx.moveTo(x, y + T); ctx.lineTo(x + T/2, y + 2); ctx.lineTo(x + T, y + T);
          ctx.fill();
          ctx.fillStyle = '#f88';
          ctx.beginPath();
          ctx.moveTo(x + 3, y + T); ctx.lineTo(x + T/2, y + 5); ctx.lineTo(x + T - 3, y + T);
          ctx.fill();
        }
        if (tile === TILE.EXIT) {
          // Glowing exit
          ctx.fillStyle = `rgba(255,255,0,${0.3 + Math.sin(frameCount * 0.1) * 0.2})`;
          ctx.fillRect(x - 2, y - 2, T + 4, T + 4);
          ctx.fillStyle = '#ff0';
          ctx.fillRect(x, y, T, T);
          ctx.fillStyle = '#fff';
          ctx.fillRect(x + 4, y + 4, T - 8, T - 8);
        }
      }
      else if (tile === TILE.COIN) {
        const bob = Math.sin(frameCount * 0.08 + tx) * 3;
        ctx.fillStyle = '#fc4';
        ctx.beginPath();
        ctx.arc(x + T/2, y + T/2 + bob, 5, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillStyle = '#fe8';
        ctx.beginPath();
        ctx.arc(x + T/2 - 1, y + T/2 + bob - 1, 2, 0, Math.PI * 2);
        ctx.fill();
      }
      else if (tile === TILE.GEM) {
        const bob = Math.sin(frameCount * 0.06 + tx * 2) * 3;
        ctx.fillStyle = '#f4f';
        ctx.save();
        ctx.translate(x + T/2, y + T/2 + bob);
        ctx.rotate(frameCount * 0.03);
        ctx.fillRect(-4, -4, 8, 8);
        ctx.fillStyle = '#faf';
        ctx.fillRect(-2, -2, 4, 4);
        ctx.restore();
      }
      else if (tile >= TILE.POWERUP_SPEED && tile <= TILE.POWERUP_MAGNET) {
        const bob = Math.sin(frameCount * 0.07 + tx) * 4;
        const colors = { [TILE.POWERUP_SPEED]: '#ff4', [TILE.POWERUP_SHIELD]: '#4cf', [TILE.POWERUP_MAGNET]: '#f84' };
        const icons = { [TILE.POWERUP_SPEED]: '⚡', [TILE.POWERUP_SHIELD]: '🛡', [TILE.POWERUP_MAGNET]: '🧲' };
        ctx.fillStyle = colors[tile];
        ctx.globalAlpha = 0.6 + Math.sin(frameCount * 0.1) * 0.3;
        ctx.fillRect(x + 2, y + 2 + bob, T - 4, T - 4);
        ctx.globalAlpha = 1;
        ctx.fillStyle = '#fff';
        ctx.font = '10px monospace';
        ctx.fillText(icons[tile] || '?', x + 2, y + T - 3 + bob);
      }
    }
  }

  // ── Moving platforms ──
  for (const mp of gs.level.movingPlatforms) {
    ctx.fillStyle = '#7a5';
    ctx.fillRect(mp.x, mp.y, mp.w, mp.h);
    ctx.fillStyle = '#685';
    ctx.fillRect(mp.x, mp.y + mp.h - 3, mp.w, 3);
    // Track dots
    ctx.fillStyle = '#5a4';
    for (let i = 0; i < mp.w; i += 8) {
      ctx.fillRect(mp.x + i + 2, mp.y + 2, 3, 2);
    }
  }

  // ── Enemies ──
  for (const e of gs.entities) {
    if (!e.alive) continue;
    if (e.type === 'walker') {
      // Simple pixel enemy
      const ec = [
        [0,'#f44','#f44',0],
        ['#d22','#fff','#fff','#d22'],
        ['#f44','#f44','#f44','#f44'],
        ['#d22',0,0,'#d22'],
      ];
      drawSprite(ec, e.x, e.y, 4, e.vx < 0);
    } else {
      // Flyer
      const wingPhase = Math.sin(e.frame * 0.3) > 0;
      const fc = [
        [wingPhase?'#a4f':0,'#84d','#84d',wingPhase?'#a4f':0],
        ['#a4f','#fff','#fff','#a4f'],
        [0,'#84d','#84d',0],
        [wingPhase?0:'#a4f',0,0,wingPhase?0:'#a4f'],
      ];
      drawSprite(fc, e.x, e.y, 4, e.vx < 0);
    }
  }

  // ── Player ──
  const p = gs.player;
  if (p.alive || frameCount % 4 < 2) {
    if (p.invincible > 0 && frameCount % 4 < 2) {
      // Flash when invincible
    } else {
      const spriteFrame = p.onGround ? (Math.abs(p.vx) > 0.5 ? p.frame : 0) : 2;
      const sprite = makePlayerSprite(p.skin.colors, spriteFrame);
      drawSprite(sprite, p.x - 1, p.y, 1, p.facing < 0);

      // Shield visual
      if (p.shieldHP > 0) {
        ctx.strokeStyle = '#4cf';
        ctx.lineWidth = 1;
        ctx.globalAlpha = 0.5 + Math.sin(frameCount * 0.1) * 0.2;
        ctx.beginPath();
        ctx.arc(p.x + p.w/2, p.y + p.h/2, 12, 0, Math.PI * 2);
        ctx.stroke();
        ctx.globalAlpha = 1;
      }

      // Dash trail
      if (p.dashTimer > 0) {
        ctx.globalAlpha = 0.3;
        drawSprite(sprite, p.x - p.dashDir * 8, p.y, 1, p.facing < 0);
        drawSprite(sprite, p.x - p.dashDir * 16, p.y, 1, p.facing < 0);
        ctx.globalAlpha = 1;
      }
    }
  }

  // ── Particles ──
  for (const part of gs.particles) {
    ctx.globalAlpha = part.life / part.maxLife;
    ctx.fillStyle = part.color;
    ctx.fillRect(part.x - part.size/2, part.y - part.size/2, part.size, part.size);
  }
  ctx.globalAlpha = 1;

  // ── Floating texts ──
  for (const ft of gs.floatingTexts) {
    ctx.globalAlpha = ft.life / ft.maxLife;
    ctx.fillStyle = ft.color;
    ctx.font = 'bold 10px monospace';
    ctx.fillText(ft.text, ft.x, ft.y);
  }
  ctx.globalAlpha = 1;

  ctx.restore();

  // ── Power-up indicators ──
  let indicatorY = CONFIG.CANVAS_HEIGHT - 20;
  if (gs.powerUps.speed > 0) { drawPowerUpIndicator('SPD', '#ff4', gs.powerUps.speed / 480, 10, indicatorY); indicatorY -= 14; }
  if (gs.powerUps.shield > 0) { drawPowerUpIndicator('SHD', '#4cf', gs.powerUps.shield / 600, 10, indicatorY); indicatorY -= 14; }
  if (gs.powerUps.magnet > 0) { drawPowerUpIndicator('MAG', '#f84', gs.powerUps.magnet / 480, 10, indicatorY); indicatorY -= 14; }
}

function drawPowerUpIndicator(label, color, pct, x, y) {

  ctx.fillRect(x, y, 80 * pct, 10);
  ctx.fillStyle = '#fff';
  ctx.font = '8px monospace';
  ctx.fillText(label, x + 2, y + 8);
}

let lastTime = 0;
let accumulator = 0;
const TICK = 1000 / CONFIG.FPS;

function gameLoop(timestamp) {
  if (!lastTime) lastTime = timestamp;
  let delta = timestamp - lastTime;
  lastTime = timestamp;

  // Cap accumulator to prevent spiral of death
  accumulator += Math.min(delta, 100);

  while (accumulator >= TICK) {
    update();
    accumulator -= TICK;
  }

  render();
  requestAnimationFrame(gameLoop);
}
